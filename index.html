<script>
// UNIFIED STATE ENGINE
const STATE_KEY = 'batcomputer_state_v1';
let appState = {
level: 1,
xp: 0,
tasks: { phy: false, chem: false, bio: false, autopsy: false },
activeTimerEnd: null,
timerLabel: 'STANDBY PROTOCOL'
};

function loadState() {
const saved = localStorage.getItem(STATE_KEY);
if (saved) {
try { appState = JSON.parse(saved); } catch(e){}
}
renderUI();
}

function saveState() {
localStorage.setItem(STATE_KEY, JSON.stringify(appState));
renderUI();
}

function renderUI() {
document.getElementById('levelBadge').innerText = `LVL ${appState.level}: INITIATE`;
for (let k in appState.tasks) {
const card = document.getElementById(`task-${k}`);
if (card) {
if (appState.tasks[k]) card.classList.add('done');
else card.classList.remove('done');
}
}
}

function toggleTask(id) {
appState.tasks[id] = !appState.tasks[id];
if (appState.tasks[id]) {
speakFriday(`Target complete. Progress logged to central database.`);
}
saveState();
}

// FRIDAY VOICE ENGINE (WEB SPEECH API)
const synth = window.speechSynthesis;
let recognition = null;
const voiceStatus = document.getElementById('voiceStatus');
const arcReactor = document.getElementById('arcReactor');

function speakFriday(text) {
if (!synth) return;
synth.cancel();
const utter = new SpeechSynthesisUtterance(text);
utter.rate = 1.05;
utter.pitch = 0.95;

utter.onstart = () => {
arcReactor.classList.add('active');
voiceStatus.innerText = `FRIDAY: "${text}"`;
};
utter.onend = () => {
arcReactor.classList.remove('active');
};
synth.speak(utter);
}

document.getElementById('briefingBtn').addEventListener('click', () => {
speakFriday("All systems operational, Boss. Target: 650 plus. Physics sprint standing by. Ready when you are.");
});

// VOICE RECOGNITION (HANDS-FREE HOOK)
const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;
if (SpeechRec) {
recognition = new SpeechRec();
recognition.continuous = false;
recognition.lang = 'en-US';

recognition.onstart = () => {
voiceStatus.innerText = "Listening for audio commands...";
arcReactor.classList.add('active');
};

recognition.onresult = (e) => {
const cmd = e.results[0][0].transcript.toLowerCase();
voiceStatus.innerText = `Heard: "${cmd}"`;
handleVoiceCommand(cmd);
};

recognition.onend = () => {
arcReactor.classList.remove('active');
};

document.getElementById('listenBtn').addEventListener('click', () => {
try { recognition.start(); } catch(e){}
});
} else {
document.getElementById('listenBtn').style.display = 'none';
}

function handleVoiceCommand(cmd) {
if (cmd.includes('physics')) {
startTimer(45, 'PHYSICS SPRINT');
speakFriday('Starting 45 minute Physics drill.');
} else if (cmd.includes('break')) {
startTimer(45, 'RECOVERY BREAK');
speakFriday('45 minute recovery timer initiated. Take your break, Boss.');
} else if (cmd.includes('status')) {
const pending = Object.values(appState.tasks).filter(v => !v).length;
speakFriday(`Status check: You have ${pending} core duties remaining for today.`);
} else {
speakFriday(`Command logged: ${cmd}. Processing.`);
}
}

// TIMESTAMP CLOUD-SYNCABLE TIMER
let timerInterval = null;

function startTimer(minutes, label) {
appState.timerLabel = label;
appState.activeTimerEnd = Date.now() + (minutes * 60 * 1000);
saveState();
runTimerLoop();
}

function resetTimer() {
appState.activeTimerEnd = null;
appState.timerLabel = 'STANDBY PROTOCOL';
saveState();
if (timerInterval) clearInterval(timerInterval);
document.getElementById('clockDisplay').innerText = "00:00";
document.getElementById('timerLabel').innerText = appState.timerLabel;
}

function runTimerLoop() {
if (timerInterval) clearInterval(timerInterval);
timerInterval = setInterval(() => {
if (!appState.activeTimerEnd) {
clearInterval(timerInterval);
return;
}
const remaining = Math.max(0, Math.floor((appState.activeTimerEnd - Date.now()) / 1000));
document.getElementById('timerLabel').innerText = appState.timerLabel;

const m = Math.floor(remaining / 60).toString().padStart(2, '0');
const s = (remaining % 60).toString().padStart(2, '0');
document.getElementById('clockDisplay').innerText = `${m}:${s}`;

if (remaining <= 0) {
clearInterval(timerInterval);
appState.activeTimerEnd = null;
saveState();
speakFriday(`${appState.timerLabel} concluded. Return to station.`);
}
}, 1000);
}

function exportState() {
const raw = JSON.stringify(appState);
navigator.clipboard.writeText(raw).then(() => {
alert("State copied to clipboard! Paste it into your laptop anytime.");
});
}

// BOOT
loadState();
if (appState.activeTimerEnd && appState.activeTimerEnd > Date.now()) {
runTimerLoop();
}
